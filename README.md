CREATE OR REPLACE PROCEDURE INSERTAR_OPERACION (
    P_NUMERO1    IN NUMBER,
    P_NUMERO2    IN NUMBER,
    P_OPERADOR   IN VARCHAR2,
    P_RESULTADO  IN NUMBER
)
AS
BEGIN

    INSERT INTO OPERACIONES_CALCULADORA
    (
        NUMERO1,
        NUMERO2,
        OPERADOR,
        RESULTADO
    )
    VALUES
    (
        P_NUMERO1,
        P_NUMERO2,
        P_OPERADOR,
        P_RESULTADO
    );

    COMMIT;

END;

using System.Data;
using Oracle.ManagedDataAccess.Client;
using CalculadoraOracle19C.Data;

namespace CalculadoraOracle19C.Forms
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void Form1_Load(object sender, EventArgs e)
        {
            lblResultado.Text = "Resultado: 0";
        }

        private bool ObtenerNumeros(
            out decimal numero1,
            out decimal numero2)
        {
            numero1 = 0;
            numero2 = 0;

            if (!decimal.TryParse(
                    txtNumero1.Text,
                    out numero1))
            {
                MessageBox.Show(
                    "Ingrese un número válido en Número 1.",
                    "Dato inválido",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning);

                txtNumero1.Focus();
                return false;
            }

            if (!decimal.TryParse(
                    txtNumero2.Text,
                    out numero2))
            {
                MessageBox.Show(
                    "Ingrese un número válido en Número 2.",
                    "Dato inválido",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning);

                txtNumero2.Focus();
                return false;
            }

            return true;
        }

        private void RealizarOperacion(string operador)
        {
            if (!ObtenerNumeros(
                    out decimal numero1,
                    out decimal numero2))
            {
                return;
            }

            decimal resultado;

            try
            {
                switch (operador)
                {
                    case "+":
                        resultado = numero1 + numero2;
                        break;

                    case "-":
                        resultado = numero1 - numero2;
                        break;

                    case "*":
                        resultado = numero1 * numero2;
                        break;

                    case "/":

                        if (numero2 == 0)
                        {
                            MessageBox.Show(
                                "No es posible dividir entre cero.",
                                "Operación inválida",
                                MessageBoxButtons.OK,
                                MessageBoxIcon.Warning);

                            return;
                        }

                        resultado = numero1 / numero2;
                        break;

                    default:
                        MessageBox.Show(
                            "Operador no válido.",
                            "Error",
                            MessageBoxButtons.OK,
                            MessageBoxIcon.Error);

                        return;
                }

                lblResultado.Text =
                    $"Resultado: {resultado:N4}";

                GuardarOperacion(
                    numero1,
                    numero2,
                    operador,
                    resultado);
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Ocurrió un error al realizar la operación:\n\n"
                    + ex.Message,
                    "Error",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Error);
            }
        }

        private void GuardarOperacion(
            decimal numero1,
            decimal numero2,
            string operador,
            decimal resultado)
        {
            try
            {
                using OracleConnection conexion =
                    ConexionOracle.ObtenerConexion();

                using OracleCommand comando =
                    new OracleCommand(
                        "INSERTAR_OPERACION",
                        conexion);

                comando.CommandType =
                    CommandType.StoredProcedure;

                comando.Parameters.Add(
                    "P_NUMERO1",
                    OracleDbType.Decimal).Value =
                    numero1;

                comando.Parameters.Add(
                    "P_NUMERO2",
                    OracleDbType.Decimal).Value =
                    numero2;

                comando.Parameters.Add(
                    "P_OPERADOR",
                    OracleDbType.Varchar2).Value =
                    operador;

                comando.Parameters.Add(
                    "P_RESULTADO",
                    OracleDbType.Decimal).Value =
                    resultado;

                conexion.Open();

                comando.ExecuteNonQuery();

                MessageBox.Show(
                    "Operación guardada correctamente en Oracle 19c.",
                    "Información",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Information);
            }
            catch (OracleException ex)
            {
                MessageBox.Show(
                    "No fue posible guardar la operación en Oracle:\n\n"
                    + ex.Message,
                    "Error Oracle",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Error);
            }
        }

        private void btnSumar_Click(object sender, EventArgs e)
        {
            RealizarOperacion("+");
        }

        private void btnRestar_Click(object sender, EventArgs e)
        {
            RealizarOperacion("-");
        }

        private void btnMultiplicar_Click(object sender, EventArgs e)
        {
            RealizarOperacion("*");
        }

        private void btnDividir_Click(object sender, EventArgs e)
        {
            RealizarOperacion("/");
        }

        private void btnHistorial_Click(object sender, EventArgs e)
        {
            CargarHistorial();
        }

        private void CargarHistorial()
        {
            try
            {
                using OracleConnection conexion =
                    ConexionOracle.ObtenerConexion();

                string sql = @"
                    SELECT
                        ID,
                        NUMERO1,
                        NUMERO2,
                        OPERADOR,
                        RESULTADO,
                        FECHA_OPERACION
                    FROM OPERACIONES_CALCULADORA
                    ORDER BY ID DESC";

                using OracleCommand comando =
                    new OracleCommand(sql, conexion);

                using OracleDataAdapter adapter =
                    new OracleDataAdapter(comando);

                DataTable tabla = new DataTable();

                adapter.Fill(tabla);

                dgvHistorial.DataSource = tabla;

                if (dgvHistorial.Columns.Count > 0)
                {
                    dgvHistorial.Columns["ID"].HeaderText =
                        "ID";

                    dgvHistorial.Columns["NUMERO1"].HeaderText =
                        "Número 1";

                    dgvHistorial.Columns["NUMERO2"].HeaderText =
                        "Número 2";

                    dgvHistorial.Columns["OPERADOR"].HeaderText =
                        "Operador";

                    dgvHistorial.Columns["RESULTADO"].HeaderText =
                        "Resultado";

                    dgvHistorial.Columns["FECHA_OPERACION"].HeaderText =
                        "Fecha";
                }
            }
            catch (OracleException ex)
            {
                MessageBox.Show(
                    "No fue posible consultar Oracle:\n\n"
                    + ex.Message,
                    "Error Oracle",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Error);
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Ocurrió un error:\n\n"
                    + ex.Message,
                    "Error",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Error);
            }
        }
    }
}

27. Notas importantes

No es necesario instalar Oracle Client tradicional para utilizar:

Oracle.ManagedDataAccess

porque es un proveedor administrado para .NET.

La cadena de conexión es el elemento que normalmente deberá adaptarse al entorno local.

Ejemplo:

"User Id=system;" +
"Password=oracle;" +
"Data Source=localhost:1521/XEPDB1;";

Para un entorno real se recomienda no dejar las credenciales directamente dentro del código fuente.
